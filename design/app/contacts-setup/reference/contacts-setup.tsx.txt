/* DIDI-744 v3: direction B, one row in the "Your card is live" dialog.
   Adding the contact happens through Google (People API contacts scope, incremental auth on tap).
   The server creates ONE "We Met" contact holding every sender address (or adds a missing one).
   Until Google verification only test users see the row; everyone else gets the dialog without it.
   If consent is declined or blocked by a Workspace admin, the row hides quietly (no error, no fallback).
   Sender addresses are served by the app; never hardcode them in product code. */

type State = "baseline" | "default" | "working" | "success" | "hidden" | "reappear";

const NEW_ADDRESS = "hello@send.we-met.link"; // example of a sender added later, server side
const CARD_URL = "we-met.link/hi/maya-chen";

const STATES: { id: State; label: string; desc: string }[] = [
  { id: "baseline", label: "Baseline", desc: "Today: Your card is live with the link, Copy and Open card. No contacts step." },
  { id: "default", label: "Default", desc: "Google test users only until verification: one row with a Continue with Google button. Google's consent screen opens on tap (incremental authorization, contacts scope)." },
  { id: "working", label: "Working", desc: "Back from Google: the server finds or creates one We Met contact with both addresses." },
  { id: "success", label: "Success", desc: "You're all set. Close and reopen the dialog to see the next open." },
  { id: "hidden", label: "Next open", desc: "No row. Shown after success, after consent is declined or blocked by a Workspace admin (the row hides quietly), and to everyone who is not a Google test user until verification. The row only returns if a new sender address is added server side." },
  { id: "reappear", label: "Reappear", desc: "A new sender address was added server side. The row returns with one button to add just that address to the We Met contact." },
];

/* ---------- Small parts ---------- */

function useCopy() {
  const [copied, setCopied] = useState(false);
  return {
    copied,
    copy: (v: string) => {
      try { void navigator.clipboard?.writeText(v); } catch { /* preview only */ }
      setCopied(true);
      window.setTimeout(() => setCopied(false), 1500);
    },
  };
}

function LinkCopyRow({ value }: { value: string }) {
  const c = useCopy();
  return (
    <div className="flex min-h-[52px] items-center gap-2 rounded-[12px] bg-[var(--surface-2)] py-1 pl-4 pr-1.5 text-[16px] font-medium leading-tight">
      <span className="min-w-0 flex-1 truncate">{value}</span>
      <button type="button" onClick={() => c.copy(value)} className={`${btnPrimary} min-h-11 px-[18px] text-[15px]`}>
        {c.copied ? <CheckIcon size={18} /> : <CopyIcon size={18} />}
        <span>{c.copied ? "Copied" : "Copy"}</span>
      </button>
    </div>
  );
}

/* Plain "Continue with Google" button per Google's sign-in button guidelines:
   standard G mark, Roboto Medium 14px, pill shape, light #FFF/#747775/#1F1F1F, dark #131314/#8E918F/#E3E3E3. */
function GoogleButton({ mode, onClick }: { mode: Mode; onClick: () => void }) {
  const c = mode === "dark" ? { bg: "#131314", border: "#8E918F", text: "#E3E3E3" } : { bg: "#FFFFFF", border: "#747775", text: "#1F1F1F" };
  return (
    <button
      type="button"
      onClick={onClick}
      className="group relative inline-flex h-10 w-full items-center justify-center gap-2.5 overflow-hidden rounded-full border px-3 text-[14px] font-medium leading-5 tracking-[0.25px] focus-visible:outline focus-visible:outline-[3px] focus-visible:outline-offset-[3px] focus-visible:outline-[var(--accent)]"
      style={{ background: c.bg, borderColor: c.border, color: c.text, fontFamily: '"Roboto", "Figtree", system-ui, sans-serif' }}
    >
      <span aria-hidden className="pointer-events-none absolute inset-0 bg-current opacity-0 transition-opacity group-hover:opacity-[0.08] group-active:opacity-[0.12]" />
      <GoogleG />
      <span className="relative">Continue with Google</span>
    </button>
  );
}

function Card({ children, live }: { children: ReactNode; live?: boolean }) {
  return (
    <section

/* ---------- The row, by state ---------- */

type RowProps = { state: State; mode: Mode; onGoogle: () => void };

function ContactsRow({ state, mode, onGoogle }: RowProps) {
  if (state === "default") {
    return (
      <Card>
        <RowHead icon={<ContactIcon size={20} />} title="Add We Met to my contacts" text="Follow-ups from people you meet come from We Met. Saving us as one contact keeps them out of Spam." />
        <GoogleButton mode={mode} onClick={onGoogle} />
        <p className="-mt-1 text-center text-[12px] leading-[1.45] text-[var(--ink-muted)]">Google asks once. We only create or update the We Met contact.</p>
      </Card>
    );
  }
  if (state === "working") {
    return (
      <Card live>
        <div role="status" className="flex items-start gap-3">
          <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-[var(--surface-2)]"><Spinner /></span>
          <div className="flex min-w-0 flex-1 flex-col gap-0.5 pt-0.5">
            <h3 className="text-[16px] font-semibold leading-snug">Adding We Met to your contacts</h3>
            <p className="text-[14px] leading-[1.45] text-[var(--ink-muted)]">Saving both sending addresses to one contact. This takes a few seconds.</p>
          </div>
        </div>
        <div aria-hidden className="h-1 overflow-hidden rounded-full bg-[var(--surface-2)]">
          <div className="wm-progress h-full w-1/3 rounded-full bg-[var(--ink)]" />
        </div>
      </Card>
    );
  }
  if (state === "success") {
    return (
      <Card live>
        <div role="status">
          <RowHead
            filled
            icon={<CheckIcon size={20} />}
            title="You're all set"
            text="We Met is in your Google Contacts with both addresses. This step won't show again."
          />
        </div>
      </Card>
    );
  }
  if (state === "reappear") {
    return (
      <Card>
        <RowHead
          icon={<ContactIcon size={20} />}
          title="We added a new sending address"
          text={<>Follow-ups can now also come from <b>{NEW_ADDRESS}</b>. Add it to your We Met contact.</>}
        />
        <button type="button" onClick={onGoogle} className={`${btnPrimary} min-h-11 w-full text-[15px]`}>Add to my We Met contact</button>
      </Card>
    );
  }
  return null;
}

/* ---------- Root ---------- */

export default function Didi744() {
  const [state, setState] = useState<State>(param("s", "default") as State);
  const [mode, setMode] = useState<Mode>(param("theme", "light") as Mode);
  const [auto, setAuto] = useState(false);
  const [open, setOpen] = useState(true);
  const capture = param("capture", "0") === "1";
  const current = STATES.find((s) => s.id === state) ?? STATES[1];

  // Prototype only: after "returning from consent", Working resolves to Success.
  useEffect(() => {
    if (state !== "working" || !auto) return;
    const t = window.setTimeout(() => { setAuto(false); setState("success"); }, 1800);
    return () => window.clearTimeout(t);
  }, [state, auto]);

  const pick = (s: State) => { setAuto(false); setOpen(true); setState(s); };
  const onGoogle = () => { setAuto(true); setState("working"); };
  const onClose = () => { setOpen(false); if (state === "success") setState("hidden"); };

  return (
    <ThemeRoot mode={mode}>
      <Switcher title="DIDI-744 v3" hidden={capture}>
        <div data-switcher className="contents">
          <Seg label="State" value={state} onChange={pick} options={STATES.map((s) => ({ id: s.id, label: s.label }))} />
          <Seg label="Theme" value={mode} onChange={setMode} options={[{ id: "light", label: "Light" }, { id: "dark", label: "Dark" }]} />
          <p className="basis-full text-[12px] leading-snug text-[var(--ink-muted)]">{current.desc} Past 900px wide the sheet becomes a centered modal.</p>
        </div>
      </Switcher>
      <EditorShell onLive={() => setOpen(true)} />
      {open && (
        <Dialog title="Your card is live" onClose={onClose}>
          <p className="text-[16px] leading-[1.55] text-[var(--ink-muted)]">Share this link, or open your card to show its QR code.</p>
          <LinkCopyRow value={CARD_URL} />
          <ContactsRow key={state} state={state} mode={mode} onGoogle={onGoogle} />
          <div className="mt-1 flex flex-col gap-2.5">
            <a href="#" onClick={(e) => e.preventDefault()} className={`${btnOutline} min-h-14 w-full text-[17px] no-underline`}>
              <ExternalIcon />Open card
            </a>
          </div>
        </Dialog>
      )}
    </ThemeRoot>
  );
}
