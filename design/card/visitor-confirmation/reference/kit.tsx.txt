export function param(name: string, fallback: string): string {
  if (typeof window === "undefined") return fallback;
  return new URLSearchParams(window.location.search).get(name) ?? fallback;
}

export function ThemeRoot({ mode, children, className = "" }: { mode: Mode; children: ReactNode; className?: string }) {
  return (
    <div
      data-theme={mode}
      className={`wm-root min-h-screen w-full bg-[var(--bg)] text-[var(--ink)] ${className}`}
      style={{ ...(TOKENS[mode] as CSSProperties), colorScheme: mode }}
    >
      {children}
    </div>
  );
}

export type Opt<T extends string> = { id: T; label: string };

export function Seg<T extends string>({ label, value, options, onChange }: { label: string; value: T; options: Opt<T>[]; onChange: (v: T) => void }) {
  return (
    <div role="radiogroup" aria-label={label} className="flex items-center gap-1 rounded-full border border-[var(--line)] bg-[var(--surface)] p-1">
      {options.map((o) => (
        <button
          key={o.id}
          type="button"
          role="radio"
          aria-checked={value === o.id}
          onClick={() => onChange(o.id)}
          className={`min-h-8 rounded-full px-3 text-[13px] font-semibold transition-colors focus-visible:outline focus-visible:outline-2 focus-visible:outline-[var(--accent)] ${
            value === o.id ? "bg-[var(--accent)] text-[var(--accent-ink)]" : "text-[var(--ink-muted)] hover:bg-[var(--surface-2)] hover:text-[var(--ink)]"
          }`}
        >
          {o.label}
        </button>
      ))}
    </div>
  );
}

/** Small labelled design switcher: not part of the product UI. */
export function Switcher({ title, children, hidden }: { title: string; children: ReactNode; hidden?: boolean }) {
  if (hidden) return null;
  return (
    <div className="sticky top-0 z-[60] w-full border-b border-[var(--line)] bg-[var(--bg)]/95 backdrop-blur">
      <div className="mx-auto flex max-w-[1100px] flex-wrap items-center gap-2 px-3 py-2">
        <span className="mr-1 text-[12px] font-semibold uppercase tracking-[0.08em] text-[var(--ink-muted)]">{title}</span>
        {children}
      </div>
    </div>
  );
}
