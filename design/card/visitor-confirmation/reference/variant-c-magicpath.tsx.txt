function ExchangeButton({ disabled, label }: { disabled?: boolean; label: string }) {
  return (
    <button type="button" disabled={disabled} aria-disabled={disabled} className={`${btnPrimary} min-h-14 w-full text-[17px]`}>

/* ---------- shared bits for proposals ---------- */

function useSave() {
  const [saved, setSaved] = useState(false);
  const [busy, setBusy] = useState(false);
  const save = () => {
    if (busy) return;
    setBusy(true);
    window.setTimeout(() => { setBusy(false); setSaved(true); }, 650);
  };
  return { saved, busy, save };
}

function SaveButton({ s, label = "Save contact", className = "" }: { s: ReturnType<typeof useSave>; label?: string; className?: string }) {
  return (
    <button
      type="button"
      onClick={s.save}
      aria-busy={s.busy}
      className={`${s.saved ? btnOutline : btnPrimary} min-h-14 w-full text-[17px] ${className}`}
    >
      {s.busy ? (
        <span className="h-4 w-4 animate-spin rounded-full border-2 border-current border-t-transparent" aria-hidden />
      ) : s.saved ? (
        <CheckIcon size={20} />
      ) : (
        <ContactIcon size={20} />
      )}
      {s.busy ? "Preparing contact" : s.saved ? "Saved, tap to save again" : label}
    </button>
  );
}

function Row({ icon, label, value, href, copy, compact }: { icon: ReactNode; label: string; value: string; href: string; copy?: boolean; compact?: boolean }) {
  const [copied, setCopied] = useState(false);
  return (
    <div className="flex min-h-14 items-center gap-3">
      <a href={href} onClick={(e) => e.preventDefault()} className="group flex min-w-0 flex-1 items-center gap-3 rounded-[12px] py-1.5 no-underline">
        {!compact && <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-[var(--surface-2)] text-[var(--ink)]">{icon}</span>}
        <span className="flex min-w-0 flex-col leading-tight">
          <span className="text-[13px] font-medium text-[var(--ink-muted)]">{label}</span>
          <span className="break-all text-[16px] font-medium text-[var(--ink)] group-hover:underline">{value}</span>
        </span>
      </a>
      {copy && (
        <button
          type="button"
          aria-label={`Copy ${label.toLowerCase()}`}
          onClick={() => { setCopied(true); window.setTimeout(() => setCopied(false), 1400); }}
          className="flex h-11 min-w-11 items-center justify-center gap-1.5 rounded-full px-3 text-[13px] font-semibold text-[var(--ink-muted)] transition-colors hover:bg-[var(--surface-2)] hover:text-[var(--ink)]"
        >
          {copied ? <><CheckIcon size={16} />Copied</> : <CopyIcon size={18} />}
        </button>
      )}
    </div>
  );
}

function FollowLine({ className = "" }: { className?: string }) {
  return <p className={`text-[15px] font-medium leading-normal text-[var(--ink-muted)] ${className}`}>{OWNER.first} will follow up by email.</p>;
}

function DetailsNote({ step, onAdd }: { step: Step; onAdd: () => void }) {
  if (step === "after")
    return (
      <p className="wm-fade inline-flex items-center gap-1.5 text-[14px] font-medium text-[var(--ink-muted)]">
        <CheckIcon size={16} /> Your details and note were sent too.
      </p>
    );
  return (
    <button type="button" onClick={onAdd} className="inline-flex min-h-11 items-center gap-1.5 self-start rounded-full px-1 text-[14px] font-medium text-[var(--ink-muted)] underline decoration-1 underline-offset-[3px] hover:text-[var(--ink)]">
      <NoteIcon size={16} /> Add your job title or a note
    </button>
  );
}

/* ---------- Variants ---------- */

function Baseline({ step }: { step: Step }) {
  // Today: after step 1 the thanks line shows; after details the button reads "Contact sent" (disabled).
  if (step === "reload")
    return <CardPage cta={<><ExchangeButton disabled label="Contact sent" /></>} />;
  return (
    <CardPage
      cta={
        <>
          <ExchangeButton disabled={step === "after"} label={step === "after" ? "Contact sent" : "Exchange contact"} />
          <p role="status" className="wm-fade mt-1 text-center text-[15px] font-medium leading-normal">Thanks, {VISITOR_FIRST}. Talk soon.</p>
        </>
      }
    />
  );
}

function ReloadFallback() {
  // Exchange token gone (reload / new tab / after ~2 h): today's card, with a quiet sent state.
  return (
    <CardPage
      cta={
        <>
          <ExchangeButton disabled label="Contact sent" />
          <p className="mt-1 text-center text-[14px] font-medium leading-normal text-[var(--ink-muted)]">{OWNER.first} has your details and will follow up by email.</p>
        </>
      }
    />
  );
}

function VariantA({ step, onAdd }: { step: Step; onAdd: () => void }) {
  const s = useSave();
  if (step === "reload") return <ReloadFallback />;
  return (
    <CardPage
      cta={
        <div className="wm-fade flex flex-col gap-3">
          <p role="status" className="inline-flex items-center justify-center gap-1.5 text-center text-[15px] font-semibold">
            <CheckIcon size={18} /> Contact sent to {OWNER.first}
          </p>
          <SaveButton s={s} />
          <div className="flex flex-col divide-y divide-[var(--line)]">
            <Row icon={<MailIcon />} label="Email" value={OWNER.email} href={`mailto:${OWNER.email}`} />
            <Row icon={<PhoneIcon />} label="Phone" value={OWNER.phone} href="tel:+14165550142" />
          </div>
          <div className="flex flex-col gap-1">
            <FollowLine />
            <DetailsNote step={step} onAdd={onAdd} />
          </div>
        </div>
      }
    />
  );
}

  const s = useSave();
  if (step === "reload") return <ReloadFallback />;
  return (
    <CardPage
      cta={

              <NoteIcon size={16} /> Tell {OWNER.first} more about you
            </button>
          )}
          {step === "after" && <p className="wm-fade text-center text-[14px] font-medium text-[var(--ink-muted)]">Your details were added. Thanks, {VISITOR_FIRST}.</p>}
        </div>
      }
    />
  );
}

/* ---------- Root ---------- */

export default function Didi742() {
  const [variant, setVariant] = useState<Variant>(param("v", "a") as Variant);
  const [mode, setMode] = useState<Mode>(param("theme", "light") as Mode);
  const [step, setStep] = useState<Step>(param("step", "before") as Step);
  const capture = param("capture", "0") === "1";
  const current = VARIANTS.find((v) => v.id === variant) ?? VARIANTS[0];
  const onAdd = () => setStep("after");
  return (
    <ThemeRoot mode={mode}>
      <Switcher title="DIDI-742" hidden={capture}>
        <Seg label="Design" value={variant} onChange={setVariant} options={VARIANTS.map((v) => ({ id: v.id, label: v.label }))} />
        <Seg label="State" value={step} onChange={setStep} options={[{ id: "before", label: "Before details" }, { id: "after", label: "After details" }, { id: "reload", label: "After reload" }]} />
        <Seg label="Theme" value={mode} onChange={setMode} options={[{ id: "light", label: "Light" }, { id: "dark", label: "Dark" }]} />
        <p className="basis-full text-[12px] leading-snug text-[var(--ink-muted)]">{current.desc}</p>
      </Switcher>
      <div key={`${variant}-${step}`}>
        {variant === "baseline" && <Baseline step={step} />}
        {variant === "a" && <VariantA step={step} onAdd={onAdd} />}
        {variant === "b" && <VariantB step={step} onAdd={onAdd} />}
        {variant === "c" && <VariantC step={step} onAdd={onAdd} />}
      </div>
    </ThemeRoot>
  );
}
