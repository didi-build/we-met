/* ---------- How it works (sticky panel on desktop, inline vignettes on phone) ---------- */

/* chipSlot replaces the text chip with custom content inside a fixed-height chip row (used by the direct view's Coming soon badge). */
export type Step = { title: string; body: ReactNode; chip: string; chipSlot?: ReactNode; vignette: ReactNode; extra?: ReactNode };

export function HowItWorks({ steps, after }: { steps: Step[]; after?: ReactNode }) {
  const [active, setActive] = useState(0);
  const refs = useRef<(HTMLLIElement | null)[]>([]);
  useEffect(() => {
    const els = refs.current.filter(Boolean) as HTMLLIElement[];
    if (!els.length || typeof IntersectionObserver === "undefined") return;
    const io = new IntersectionObserver(
      (entries) => {
        for (const e of entries) {
          if (e.isIntersecting) {
            const i = els.indexOf(e.target as HTMLLIElement);
            if (i >= 0) setActive(i);
          }
        }
      },
      { rootMargin: "-45% 0px -45% 0px", threshold: 0 },
    );
    els.forEach((el) => io.observe(el));
    return () => io.disconnect();
  }, []);
  return (
    <section className="lp-band" aria-labelledby="how-title">
      <div className="lp-wrap lp-how">
        <div className="lp-sechead">
          <p className="lp-eyebrow">How it works</p>
          <h2 className="lp-h2" id="how-title">Three steps, from hello to follow-up.</h2>
          <p className="lp-body">Free to start, with nothing to install on either side.</p>
        </div>
        <div className="lp-how-grid">
          <div className="lp-panel" aria-hidden="true">
            {steps.map((s, i) => (
              <div key={i} className="lp-layer" data-active={i === active ? "true" : "false"}>
                {s.vignette}
              </div>
            ))}
          </div>
          <ol className="lp-steps">
            {steps.map((s, i) => (
              <li
                key={i}
                className="lp-step"
                data-active={i === active ? "true" : "false"}
                ref={(el) => {
                  refs.current[i] = el;
                }}
                onMouseEnter={() => setActive(i)}
              >
                <span className="lp-num" aria-hidden="true">{i + 1}</span>
                <h3>{s.title}</h3>
                <p className="lp-body">{s.body}</p>
                {s.chipSlot !== undefined ? (
                  <ul className="lp-chips"><li className="lp-chip-slot">{s.chipSlot}</li></ul>
                ) : (
                  <ul className="lp-chips"><li><span className="lp-chip">{s.chip}</span></li></ul>
                )}
                {s.extra}
                <div className="lp-vig-box" aria-hidden="true">{s.vignette}</div>
              </li>
            ))}
          </ol>
        </div>
        {after}
      </div>
    </section>
  );
}
