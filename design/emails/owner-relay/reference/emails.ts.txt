const esc = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");

/** Minimal handlebars: {{#if x}}..{{else}}..{{/if}} (non-nested) and {{var}}. */
export function fill(tpl: string, vars: Record<string, string | undefined>, html = true): string {
  const out = tpl.replace(/\{\{#if (\w+)\}\}([\s\S]*?)\{\{\/if\}\}/g, (_m, k: string, body: string) => {
    const [yes, no = ""] = body.split("{{else}}");
    return vars[k] ? yes : no;
  });
  return out.replace(/\{\{(\w+)\}\}/g, (_m, k: string) => {
    const v = vars[k] ?? "";
    return html && !k.endsWith("_url") ? esc(v) : v;
  });
}

const darkify = (html: string, dark: boolean) =>
  dark ? html.replace("@media (prefers-color-scheme:dark){", "@media all{") : html;

export function renderVisitor(v: VisitorVariant, o: { test?: boolean; dark?: boolean } = {}): string {
  return darkify(fill(TEMPLATES[`visitor-followup.${v}.html`], visitorVars(!!o.test)), !!o.dark);
}
export function renderVisitorText(v: VisitorVariant, test = false): string {
  return fill(TEMPLATES[`visitor-followup.${v}.txt`], visitorVars(test), false);
}
export function renderDetails(v: DetailsVariant, o: { test?: boolean; dark?: boolean } = {}): string {
  return darkify(fill(TEMPLATES[`owner-details-update.${v}.html`], detailsVars(!!o.test)), !!o.dark);
}
export function renderDetailsText(v: DetailsVariant, test = false): string {

export function subjectFor(subject: string, test: boolean, to: string): string {
  return test ? `[we-met test → ${to}] ${subject}` : subject;
}

export function renderOwner(v: OwnerVariant, o: { test?: boolean; dark?: boolean } = {}): string {
  const tpl = v === "baseline" ? TEMPLATES["owner-new-lead.baseline.html"] : TEMPLATES["owner-relay.html"];
  // Baseline = today's New lead email, as sent today (phone is still on today's exchange form).
  const vars = ownerVars(!!o.test, v !== "minimal");
  return darkify(fill(tpl, vars), !!o.dark);
}
export function renderOwnerText(v: OwnerVariant, test = false): string {
  const tpl = v === "baseline" ? TEMPLATES["owner-new-lead.baseline.txt"] : TEMPLATES["owner-relay.txt"];
  return fill(tpl, ownerVars(test, v !== "minimal"), false);
}
