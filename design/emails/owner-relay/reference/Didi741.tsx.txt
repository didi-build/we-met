function EmailFrame({
  html,
  title
}: {
  html: string;
  title: string;
}) {
  const ref = useRef<HTMLIFrameElement>(null);
  const [h, setH] = useState(600);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const fit = () => {
      const body = el.contentDocument?.body;
      if (!body) return;
      const bottoms = Array.from(body.children).map(c => (c as HTMLElement).getBoundingClientRect().bottom);
      if (bottoms.length) setH(Math.max(120, Math.ceil(Math.max(...bottoms))));
    };
    el.addEventListener("load", fit);
    const id = window.setInterval(fit, 400);
    return () => {
      el.removeEventListener("load", fit);
      window.clearInterval(id);
    };
  }, [html]);
  return <iframe ref={ref} title={title} srcDoc={html} className="block w-full border-0" style={{
    height: h
  }} />;
}
